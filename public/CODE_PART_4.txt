                                              if (chapter.children) {
                                                return { ...chapter, children: update(chapter.children) };

                                          return newChapters;

                                        return 100;

                                      return prev + 10;
                                    });
                                  }, 150);

                            className="flex items-center gap-1 px-3 py-1.5 text-sm bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors"

  const [knowledgeSearchKeyword, setKnowledgeSearchKeyword] = useState('');

  // 发布相关状态

  const [showHistoryModal, setShowHistoryModal] = useState(false);

    })
  );

  // 加载页面状态
  useEffect(() => {
    if (selectedTopic) {
      const state = mockPageState[selectedTopic.id];
      if (state) {
        setPageState(state);
        if (state.pendingChanges > 0) {
          const changes: ChangeRecord[] = [
            { type: 'modify', target: '函数基础', detail: '修改知识点配置' },
            { type: 'add', target: '指数函数', detail: '新增知识点' },
          ];
          setPendingChanges(changes.slice(0, state.pendingChanges));
        } else {
          setPendingChanges([]);
        }
      }
    }
  }, [selectedTopic]);

  // 筛选专题卡片
  const filteredTopicCards = useMemo(() => {
    return mockTopicCards.filter(card => {
      const phaseMatch = !selectedPhase || card.phase === phaseConfig[selectedPhase].label;
      const subjectMatch = !selectedSubject || subjectConfig.find(s => s.id === selectedSubject)?.name === card.subject;
      const gradeMatch = !selectedGrade || gradeConfig[selectedPhase]?.find(g => g.id === selectedGrade)?.name === card.grade;
      const sceneMatch = selectedScene === 'all' || card.scene === selectedScene;
      return phaseMatch && subjectMatch && gradeMatch && sceneMatch;
    });
  }, [selectedPhase, selectedSubject, selectedGrade, selectedScene]);

  // ==================== 辅助函数 ====================

  // 生成节点编号
  const generateNodeNumber = (nodeId: string): string => {
    const parts = nodeId.split('-');
    if (parts.length === 1) {
      // 顶级节点，找到其在同级中的位置
      const index = topicTree.findIndex(n => n.id === nodeId);
      return `${index + 1}`;
    }
    // 子节点
    const parentId = parts[0];
    const parentIndex = topicTree.findIndex(n => n.id === parentId);
    const parent = topicTree[parentIndex];
    if (parent && parent.children) {
      const childIndex = parent.children.findIndex(c => c.id === nodeId);
      return `${parentIndex + 1}.${childIndex + 1}`;
    }
    return '';
  };

  // 检查节点是否有变更
  const hasNodeChanges = (node: TopicNode): boolean => {
    if (!pageState?.changedNodes) return false;
    if (pageState.changedNodes.includes(node.id)) return true;
    if (node.children) {
      for (const child of node.children) {
        if (hasNodeChanges(child)) return true;
      }
    }
    return false;
  };

  // 获取所有节点ID（用于拖拽）
  const getAllNodeIds = (nodes: TopicNode[]): string[] => {
    const ids: string[] = [];
    const traverse = (node: TopicNode) => {
      ids.push(node.id);
      if (node.children) {
        node.children.forEach(traverse);
      }
    };
    nodes.forEach(traverse);
    return ids;
  };

  // ==================== 编辑操作 ====================

  // 切换节点展开状态
  const toggleNode = (nodeId: string) => {
    const toggleRecursive = (nodes: TopicNode[]): TopicNode[] => {
      return nodes.map(node => {
        if (node.id === nodeId) {
          return { ...node, expanded: !node.expanded };
        }
        if (node.children) {
          return { ...node, children: toggleRecursive(node.children) };
        }
        return node;
      });
    };
    setTopicTree(toggleRecursive(topicTree));
  };

  // 编辑节点名称
  const handleEditNodeName = (nodeId: string, newName: string) => {
    const editRecursive = (nodes: TopicNode[]): TopicNode[] => {
      return nodes.map(node => {
        if (node.id === nodeId) {
          return { ...node, name: newName };
        }
        if (node.children) {
          return { ...node, children: editRecursive(node.children) };
        }
        return node;
      });
    };
    setTopicTree(editRecursive(topicTree));
    setEditingNodeId(null);
    setEditingNodeValue('');

    const newTopic: TopicNode = {
      id: `t${Date.now()}`,
      type: 'topic',
      name,
      order: topicTree.length + 1,
      expanded: true,
      children: [],

    setAddingParentId(null);
    setAddingValue('');
  };

  const handleAddKnowledge = (parentId: string, knowledge: KnowledgeReference) => {
    const newKnowledge: TopicNode = {
      id: `k${Date.now()}`,
      type: 'knowledge',
      name: knowledge.name,
      order: 0,
      examCount: knowledge.examCount,
      knowledgeRef: knowledge,
    };
    
    const addRecursive = (nodes: TopicNode[]): TopicNode[] => {
      return nodes.map(node => {
        if (node.id === parentId) {
          const children = node.children || [];
          return { ...node, children: [...children, { ...newKnowledge, order: children.length + 1 }] };
        }
        if (node.children) {
          return { ...node, children: addRecursive(node.children) };
        }
        return node;
      });
    };
    setTopicTree(addRecursive(topicTree));
  };

  // 删除节点
  const handleDeleteNode = (nodeId: string) => {
    const deleteRecursive = (nodes: TopicNode[]): TopicNode[] => {
      return nodes.filter(node => {
        if (node.id === nodeId) return false;
        if (node.children) {
          node.children = deleteRecursive(node.children);
        }
        return true;
      });
    };
    setTopicTree(deleteRecursive(topicTree));
    if (selectedNode?.id === nodeId) {
      setSelectedNode(null);
    }
    setShowDeleteConfirm(false);
    setDeletingNodeId(null);
  };

  // 拖拽结束
  const handleDragEnd = (event: DragEndEvent) => {
    const { active, over } = event;
    if (over && active.id !== over.id) {
      // 简化处理：只处理同级排序
      // 实际项目中需要更复杂的逻辑处理跨级拖拽
    }
  };

  // 处理发布
  const handlePublish = (version: string, description: string, scheduledInfo?: { scheduledDate: string; scheduledTime: string }) => {
    if (!pageState) return;
    
    if (scheduledInfo) {
      // 定时发布：保存定时信息

        ...pageState,

        pendingChanges: 0,

          scheduledTime: scheduledInfo.scheduledTime,
          description
        }

      setPendingChanges([]);

        publishTime: new Date().toLocaleString('zh-CN', {
          year: 'numeric',
          month: '2-digit',
          day: '2-digit',
          hour: '2-digit',
          minute: '2-digit',
        }).replace(/\//g, '-'),
        publisher: '当前用户',
      };

      setPageState({
        status: 'published',
        currentVersion: version,
        lastPublishTime: newRecord.publishTime,
        lastPublisher: newRecord.publisher,
        pendingChanges: 0,
        changedNodes: [],
        publishHistory: [newRecord, ...pageState.publishHistory],
      });
      setPendingChanges([]);
    }
  };

  // 取消定时发布
  const handleCancelScheduled = () => {
    if (!pageState || !pageState.scheduledPublish) return;
    
    // 恢复到待发布状态
    setPageState({
      ...pageState,
      status: 'pending',
      pendingChanges: 1,
      changedNodes: [],
      scheduledPublish: undefined
    });
    setPendingChanges([{ type: 'modify', target: '专题结构', detail: '取消定时发布' }]);
  };

  // 完成编辑

    }
  };

  // ==================== 子组件 ====================

  // 可排序树节点
  const SortableTreeNode = ({ id, isEditingTree }: { id: string; isEditingTree: boolean }) => {
    const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });

    const style = {
      transform: CSS.Transform.toString(transform),
      transition,
      opacity: isDragging ? 0.5 : 1,
    };

    return (
      <div ref={setNodeRef} style={style} {...attributes}>
        {/* 具体渲染在 renderTreeNode 中处理 */}
      </div>
    );

  // 渲染树节点
  const renderTreeNode = (node: TopicNode, level: number = 0): React.ReactNode => {
    const isSelected = selectedNode?.id === node.id;
    const hasChildren = node.children && node.children.length > 0;
    const paddingLeft = isEditingTree ? level * 24 + 28 : level * 24 + 12;
    const nodeNumber = generateNodeNumber(node.id);
    const isHovered = hoveredNodeId === node.id;
    const isEditingThis = editingNodeId === node.id;
    const nodeHasChanges = hasNodeChanges(node);

    return (
      <div key={node.id} className="select-none">
        <div
          className={`group flex items-center gap-2 py-2 px-2 cursor-pointer transition-colors ${
            isSelected ? 'bg-emerald-50 text-emerald-700' : 'text-gray-700 hover:bg-gray-50'
          }`}
          style={{ paddingLeft: `${paddingLeft}px` }}
          onClick={() => !isEditingThis && setSelectedNode(node)}
          onMouseEnter={() => setHoveredNodeId(node.id)}
          onMouseLeave={() => setHoveredNodeId(null)}
        >
          {/* 编辑模式：拖拽手柄 */}
          {isEditingTree && (

          {hasChildren ? (

            </button>
          ) : (

          {/* 自动编号 */}
          {isEditingTree && (
            <span className="text-xs text-gray-400 font-mono min-w-[20px]">{nodeNumber}</span>

                autoFocus
                onClick={(e) => e.stopPropagation()}
                onKeyDown={(e) => {
                  if (e.key === 'Enter' && editingNodeValue.trim()) {
                    handleEditNodeName(node.id, editingNodeValue.trim());
                  }
                  if (e.key === 'Escape') {
                    setEditingNodeId(null);
                    setEditingNodeValue('');
                  }
                }}
              />
              <button
                onClick={(e) => {
                  e.stopPropagation();
                  if (editingNodeValue.trim()) {
                    handleEditNodeName(node.id, editingNodeValue.trim());
                  }
                }}
                className="p-1 text-emerald-600 hover:bg-emerald-50 rounded"
              >
                <Check className="w-4 h-4" />
              </button>
              <button
                onClick={(e) => {
                  e.stopPropagation();
                  setEditingNodeId(null);
                  setEditingNodeValue('');
                }}
                className="p-1 text-gray-400 hover:bg-gray-50 rounded"
              >
                <X className="w-4 h-4" />
              </button>
            </div>
          ) : (
            <>
              {/* 节点名称 */}
              <span className="flex-1 font-medium truncate">
                {node.type === 'topic' ? node.name : node.name}
              </span>

              {/* 有更新标签 */}
              {!isEditingTree && nodeHasChanges && pageState?.status === 'pending' && (

                  有更新
                </span>
              )}

              {/* 知识点：考试次数 */}
              {node.type === 'knowledge' && node.examCount !== undefined && (
                <span className="text-xs px-2 py-0.5 bg-orange-100 text-orange-600 rounded-full">
                  {node.examCount}次

              {/* 知识点：引用标记 */}
              {node.type === 'knowledge' && (

                  <Link2 className="w-3 h-3" />
                  引用
                </span>
              )}

              {/* 编辑模式：操作按钮 */}
              {isEditingTree && isHovered && (
                <div className="flex items-center gap-1">

                  {node.type === 'topic' && (
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        setSelectedKnowledgeIds(new Set());
                        setKnowledgeSearchKeyword('');
                        setAddingParentId(node.id);
                        setShowKnowledgeSelector(true);
                      }}
                      className="p-1 text-gray-400 hover:text-blue-600 hover:bg-blue-50 rounded transition-colors"
                      title="添加知识点"
                    >
                      <Plus className="w-4 h-4" />
                    </button>
                  )}
                  {/* 编辑名称 */}
                  <button
                    onClick={(e) => {
                      e.stopPropagation();
                      setEditingNodeId(node.id);
                      setEditingNodeValue(node.name);
                    }}
                    className="p-1 text-gray-400 hover:text-blue-600 hover:bg-blue-50 rounded transition-colors"

                    <Edit className="w-3.5 h-3.5" />

              <History className="w-4 h-4" />
              查看历史版本

          </div>
        </div>
      );

      return (
        <div className="bg-blue-50 border border-blue-200 rounded-lg px-4 py-3 mb-6">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <Clock className="w-4 h-4 text-blue-600" />
              <span className="text-sm text-blue-800">
                <span className="font-medium">{pageState.scheduledPublish.version}</span> 版本将于您设定的时间（
                <span className="font-medium">{pageState.scheduledPublish.scheduledDate} {pageState.scheduledPublish.scheduledTime}</span>）自动发布，发布后更新的内容将正式生效
              </span>
            </div>
            <div className="flex items-center gap-2">
              <button
                onClick={() => setShowHistoryModal(true)}
                className="flex items-center gap-1.5 px-3 py-1.5 text-sm text-blue-700 hover:text-blue-800 hover:bg-blue-100 rounded-lg transition-colors"
              >
                <History className="w-4 h-4" />
                查看历史版本

              >
                取消定时
              </button>
            </div>
          </div>
        </div>
      );

    // 待发布状态（包括有定时信息但有新变更的情况）：显示橙色提醒条

      <>
        <div className="bg-amber-50 border border-amber-200 rounded-lg px-4 py-3 mb-1">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-3">
              <AlertTriangle className="w-4 h-4 text-amber-600" />

              </span>
            </div>
            <div className="flex items-center gap-2">
              <button
                onClick={() => setShowHistoryModal(true)}
                className="flex items-center gap-1.5 px-3 py-1.5 text-sm text-amber-700 hover:text-amber-800 hover:bg-amber-100 rounded-lg transition-colors"

                <History className="w-4 h-4" />
                查看历史版本

  };

    const newVersion = pageState.scheduledPublish?.version || (() => {
      const currentParts = pageState.currentVersion.replace('v', '').split('.');
      return `v${currentParts[0]}.${parseInt(currentParts[1]) + 1}.0`;
    })();

    const handleConfirmPublish = () => {
      if (!publishDescription.trim()) return;
      
      if (publishType === 'scheduled') {
        // 定时发布：传递定时信息
        handlePublish(newVersion, publishDescription, {
          scheduledDate,
          scheduledTime
        });
      } else {
        // 立即发布

  showGradeFilter?: boolean;     // 显示年级筛选

// 时间格式化工具函数
function formatPublishTime(time: string): string {
  // 如果已经是格式化的时间，直接返回
  if (time.includes('-') && time.includes(':')) {
    return time;
  }
  return time;
}

// 时间排序比较函数
function sortByPublishTime(a: AggregateHistoryItem, b: AggregateHistoryItem): number {
  const timeA = new Date(a.publishTime.replace(/-/g, '/')).getTime();
  const timeB = new Date(b.publishTime.replace(/-/g, '/')).getTime();
  return timeB - timeA; // 倒序，最新的在前
}
